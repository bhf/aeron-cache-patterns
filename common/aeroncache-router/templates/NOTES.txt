{{- if eq .Values.interface "http" }}
  curl -X PUT  localhost:8080/api/v1/cache/{{ (index .Values.routes 0).cacheId }}

  curl        localhost:8080/api/v1/cache/{{ (index .Values.routes 0).cacheId }}/k
{{- else if eq .Values.interface "sse" }}
  curl -X PUT  localhost:8080/api/v1/cache/{{ (index .Values.routes 0).cacheId }}                      # write via the routed http writer
  curl -N      localhost:8080/api/sse/v1/cache/hydrate/{{ (index .Values.routes 0).cacheId }}          # read via the routed sse gateway
{{- else if eq .Values.interface "ws" }}
  curl -X PUT  localhost:8080/api/v1/cache/{{ (index .Values.routes 0).cacheId }}                      # write via the routed http writer
  # subscribe (ws): ws://localhost:8080/api/ws/v1/cache/{{ (index .Values.routes 0).cacheId }}
{{- end }}

A cache id not in the route list has no matching path and returns 404 — routing is explicit.
